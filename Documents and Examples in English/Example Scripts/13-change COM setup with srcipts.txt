# This example shows how to open/close the serial port and change the serial port settings with a script
# Functions used in this example
# connect_serial()
# close_serial()

# set_port("COMx")
# set_baudrate("xxxx")

toggle_serial()
wait 1
set_port("COM20")
show_str("set COM20\n")
wait 1
set_baudrate("57600")
show_str("set baudrate to 57600\n")
wait 1
toggle_serial()
wait 1
close_serial()
wait 1
set_port("COM3")
show_str("set COM3\n")
wait 1
set_baudrate("115200")
show_str("set baudrate to 115200\n")
wait 1
connect_serial()
wait 1
